# Зависимости намеренно минимальны. Каждая — с обоснованием, почему
# стандартной библиотеки недостаточно.

# httpx: urllib не даёт вменяемых таймаутов (connect/read раздельно),
# ретраев и работы с условными заголовками без ручной обвязки. Ручной
# backoff на urllib — именно то место, где заводятся тихие баги
# (per-chunk timeout, который не является wall-clock лимитом).
httpx>=0.28,<0.29

# selectolax: html.parser из stdlib ломается на невалидном HTML молча —
# отдаёт неполное дерево без ошибки. Парсим чужой документ, который может
# перевёрстываться без предупреждения; устойчивость парсера здесь дороже
# отсутствия зависимости.
selectolax>=0.3.25,<0.4

# Официальный SDK. Требуется свежий: Responses API, строгие структурированные
# ответы (text.format с json_schema + strict) и hosted-инструмент web_search
# с фильтром по доменам.
openai>=2.0,<3

# Валидация структурированного ответа модели. Без неё пришлось бы вручную
# проверять двенадцать полей и вложенные объекты на каждом разборе.
pydantic>=2.9,<3
